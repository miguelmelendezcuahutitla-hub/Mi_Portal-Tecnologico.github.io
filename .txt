  <!-- Script para el menú desplegable táctil/clic y el envío de la encuesta -->
  <script>
    document.addEventListener('DOMContentLoaded', () => {
      // 1. Control de clic para el menú desplegable (útil en celulares y tablets)
      const dropdown = document.getElementById('dropdownMenu');
      const dropdownToggle = document.getElementById('dropdownToggle');

      dropdownToggle.addEventListener('click', (e) => {
        e.stopPropagation();
        const isActive = dropdown.classList.toggle('active');
        dropdownToggle.setAttribute('aria-expanded', isActive);
      });

      // Cerrar el menú desplegable si se hace clic fuera de él o en uno de sus enlaces
      document.addEventListener('click', (e) => {
        if (!dropdown.contains(e.target)) {
          dropdown.classList.remove('active');
          dropdownToggle.setAttribute('aria-expanded', 'false');
        }
      });

      dropdown.querySelectorAll('.dropdown-content a').forEach((enlace) => {
        enlace.addEventListener('click', () => {
          dropdown.classList.remove('active');
          dropdownToggle.setAttribute('aria-expanded', 'false');
        });
      });

      // 2. Manejo del formulario de encuesta sin recargar la página
      const formEncuesta = document.getElementById('formEncuesta');
      const mensajeConfirmacion = document.getElementById('mensajeConfirmacion');
      const textoConfirmacion = document.getElementById('textoConfirmacion');

      formEncuesta.addEventListener('submit', (e) => {
        e.preventDefault(); // Evita que la página se recargue

        // Obtener los valores ingresados por el usuario
        const nombre = document.getElementById('nombreVisitante').value.trim();
        const temaSeleccionado = document.querySelector('input[name="temaInteres"]:checked');
        const calificacion = document.getElementById('calificacionPortal').value;
        
        const dispositivosChecked = Array.from(
          document.querySelectorAll('input[name="dispositivos"]:checked')
        ).map(cb => cb.value);

        const temaTexto = temaSeleccionado ? temaSeleccionado.value : 'No especificado';
        const dispositivosTexto = dispositivosChecked.length > 0 
          ? dispositivosChecked.join(', ') 
          : 'Ninguno seleccionado';

        // Construir el mensaje personalizado de agradecimiento
        textoConfirmacion.innerHTML = `
          ¡Muchas gracias, <strong>${nombre}</strong>! Hemos registrado tus respuestas:<br>
          • <strong>Tema favorito:</strong> ${temaTexto}<br>
          • <strong>Calificación del portal:</strong> ${calificacion}<br>
          • <strong>Dispositivos:</strong> ${dispositivosTexto}
        `;

        // Mostrar el cuadro de confirmación y desplazarse suavemente hacia él
        mensajeConfirmacion.style.display = 'block';
        mensajeConfirmacion.scrollIntoView({ behavior: 'smooth', block: 'nearest' });

        // Limpiar el formulario después del envío
        formEncuesta.reset();
      });
    });
  </script>
